{{/* get-block.html Returns the rendered content block identified by a key. Expects: - key: string or {ref: name} - anchors: map of name → block */}} {{ $rawKey := .key }} {{ $anchors := .anchors }} {{ $key := "" }} {{ if eq (printf "%T" $rawKey) "string" }} {{ $key = $rawKey }} {{ else if and (eq (printf "%T" $rawKey) "map[string]interface {}") (index $rawKey "ref") }} {{ $key = index $rawKey "ref" }} {{ else }}
⚠️ Invalid key for block: {{ $rawKey }}
{{ end }} {{ $block := index $anchors $key }} {{ if not $block }}
⚠️ Block {{ $key }} not found in anchors.
{{ else if eq (printf "%T" $block) "string" }} {{ $block | markdownify }} {{ else if and (eq (printf "%T" $block) "map[string]interface {}") (eq (index $block "type") "paragraph") }} {{ index $block "text" | markdownify }} {{ else }}
⚠️ Block {{ $key }} has unsupported structure or missing type.
{{ end }}
